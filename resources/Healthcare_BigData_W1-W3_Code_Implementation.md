# W1–W3 Detailed Code Implementation — File-by-File

Companion to `Healthcare_BigData_Implementation_Guide.md`. Since your cluster + HDFS load (W1–W2) already works, this picks up from there: it turns Part 2 (Ingestion & Cleaning) and Part 3 (Readmission Label) of the main guide into actual files, with exact paths, full code, and the exact commands to run each one.

> **Assumption stated up front:** code below uses **MIMIC-IV Demo (v2.2)** column names (`subject_id`, `hadm_id`, `admittime`, lowercase throughout, `anchor_age` instead of DOB). If your team standardized on the **MIMIC-III Demo** instead, the logic is identical but column/table names are UPPERCASE (`SUBJECT_ID`, `ADMITTIME`, …) and there's no `anchor_age` (MIMIC-III uses `DOB` and shifts ages >89 to 300 instead of bucketing them). Everywhere that matters, there's a comment flagging the difference — just rename the `StructField`s in `table_schemas.py` and the age-capping line in `clean_patients.py`.

### `src/cleaning/data_quality_utils.py`

Shared helpers so null/duplicate checks aren't copy-pasted six times, and so every cleaning script writes to the same running report.

```python
"""
Reusable data-quality helpers used by every script in src/cleaning/.
"""
import os
from datetime import datetime
from pyspark.sql import DataFrame, functions as F


def null_report(df: DataFrame) -> dict:
    """Return {column: null_count} for every column in df."""
    row = df.select([
        F.count(F.when(F.col(c).isNull(), c)).alias(c) for c in df.columns
    ]).collect()[0]
    return row.asDict()


def duplicate_count(df: DataFrame, key_cols: list) -> int:
    """Count how many key_cols groups have more than 1 row."""
    return df.groupBy(*key_cols).count().filter("count > 1").count()


def append_report_section(report_path: str, table_name: str, rows_before: int,
                           rows_after: int, nulls: dict, dupes: int, notes: str = ""):
    """Append one markdown section to reports/cleaning_report.md."""
    os.makedirs(os.path.dirname(report_path) or ".", exist_ok=True)
    write_header = not os.path.exists(report_path)

    with open(report_path, "a") as f:
        if write_header:
            f.write("# Cleaning Report\n\n")
            f.write("Auto-generated by src/cleaning/*.py — re-run the cleaning "
                    "scripts to refresh, don't hand-edit.\n\n")
        f.write(f"## {table_name}\n")
        f.write(f"- Run at: {datetime.utcnow().isoformat()}Z\n")
        f.write(f"- Rows before cleaning: {rows_before}\n")
        f.write(f"- Rows after cleaning: {rows_after}\n")
        f.write(f"- Duplicate keys removed: {dupes}\n")
        f.write("- Null counts by column (0-count columns omitted):\n")
        for col, n in nulls.items():
            if n:
                f.write(f"  - `{col}`: {n}\n")
        if notes:
            f.write(f"- Notes: {notes}\n")
        f.write("\n")
```

### `src/cleaning/clean_admissions.py`

```python
"""
Cleans the raw `admissions` table.
Reads:  hdfs .../raw/hosp/admissions.csv.gz
Writes: hdfs .../cleaned/admissions  (Parquet)

Run:
    spark-submit --master yarn --deploy-mode client src/cleaning/clean_admissions.py
"""
from pyspark.sql import functions as F
from common.spark_session import get_spark
from common.config import PATHS, REPORTS_DIR
from schemas.table_schemas import admissions_schema
from cleaning.data_quality_utils import null_report, duplicate_count, append_report_section


def main():
    spark = get_spark("clean_admissions")

    raw = spark.read.csv(PATHS["admissions_raw"], header=True, schema=admissions_schema)
    rows_before = raw.count()
    nulls = null_report(raw)
    dupes = duplicate_count(raw, ["hadm_id"])

    clean = (
        raw
        .filter(F.col("dischtime") > F.col("admittime"))
        .withColumn(
            "los_days",
            F.round((F.col("dischtime").cast("long") - F.col("admittime").cast("long")) / 86400.0, 2),
        )
        .dropDuplicates(["hadm_id"])
    )
    rows_after = clean.count()

    clean.write.mode("overwrite").parquet(PATHS["admissions_clean"])

    append_report_section(
        f"{REPORTS_DIR}/cleaning_report.md", "admissions",
        rows_before, rows_after, nulls, dupes,
        notes="Dropped rows where dischtime <= admittime (negative/zero LOS is a data error, not a real stay).",
    )

    print(f"admissions: {rows_before} -> {rows_after} rows -> {PATHS['admissions_clean']}")
    spark.stop()


if __name__ == "__main__":
    main()
```

### `src/cleaning/clean_patients.py`

```python
"""
Cleans the raw `patients` table.

Run:
    spark-submit --master yarn --deploy-mode client src/cleaning/clean_patients.py
"""
from pyspark.sql import functions as F
from common.spark_session import get_spark
from common.config import PATHS, REPORTS_DIR
from schemas.table_schemas import patients_schema
from cleaning.data_quality_utils import null_report, duplicate_count, append_report_section


def main():
    spark = get_spark("clean_patients")

    raw = spark.read.csv(PATHS["patients_raw"], header=True, schema=patients_schema)
    rows_before = raw.count()
    nulls = null_report(raw)
    dupes = duplicate_count(raw, ["subject_id"])

    # De-identification artifact: MIMIC-III shifts ages >89 to 300; MIMIC-IV
    # instead reports ages for the very old with reduced precision. Either
    # way, flag rather than silently trust an outlier age.
    clean = (
        raw
        .withColumn("age_flagged", F.col("anchor_age") > 89)
        .withColumn("anchor_age", F.when(F.col("anchor_age") > 89, 90).otherwise(F.col("anchor_age")))
        .dropDuplicates(["subject_id"])
    )
    rows_after = clean.count()

    clean.write.mode("overwrite").parquet(PATHS["patients_clean"])

    append_report_section(
        f"{REPORTS_DIR}/cleaning_report.md", "patients",
        rows_before, rows_after, nulls, dupes,
        notes="Capped ages >89 to 90 and flagged them in `age_flagged` — de-identification artifact, not a real age.",
    )

    print(f"patients: {rows_before} -> {rows_after} rows -> {PATHS['patients_clean']}")
    spark.stop()


if __name__ == "__main__":
    main()
```

### `src/cleaning/clean_diagnoses.py`

Handles both `diagnoses_icd` and its lookup table `d_icd_diagnoses` in one job, since the lookup table needs almost no cleaning on its own.

```python
"""
Cleans `diagnoses_icd` and passes through the `d_icd_diagnoses` lookup table.

Run:
    spark-submit --master yarn --deploy-mode client src/cleaning/clean_diagnoses.py
"""
from common.spark_session import get_spark
from common.config import PATHS, REPORTS_DIR
from schemas.table_schemas import diagnoses_icd_schema, d_icd_diagnoses_schema
from cleaning.data_quality_utils import null_report, duplicate_count, append_report_section


def main():
    spark = get_spark("clean_diagnoses")

    # --- diagnoses_icd ---
    raw_dx = spark.read.csv(PATHS["diagnoses_icd_raw"], header=True, schema=diagnoses_icd_schema)
    rows_before_dx = raw_dx.count()
    nulls_dx = null_report(raw_dx)
    dupes_dx = duplicate_count(raw_dx, ["hadm_id", "seq_num"])

    clean_dx = raw_dx.dropna(subset=["icd_code", "icd_version"]).dropDuplicates(["hadm_id", "seq_num"])
    rows_after_dx = clean_dx.count()
    clean_dx.write.mode("overwrite").parquet(PATHS["diagnoses_clean"])

    append_report_section(
        f"{REPORTS_DIR}/cleaning_report.md", "diagnoses_icd",
        rows_before_dx, rows_after_dx, nulls_dx, dupes_dx,
        notes="Dropped rows missing icd_code/icd_version. Every downstream join MUST use "
              "BOTH icd_code AND icd_version — MIMIC-IV mixes ICD-9 and ICD-10 codes.",
    )

    # --- d_icd_diagnoses (reference/lookup table) ---
    raw_lookup = spark.read.csv(PATHS["d_icd_diagnoses_raw"], header=True, schema=d_icd_diagnoses_schema)
    rows_before_lk = raw_lookup.count()
    nulls_lk = null_report(raw_lookup)
    dupes_lk = duplicate_count(raw_lookup, ["icd_code", "icd_version"])

    clean_lookup = raw_lookup.dropDuplicates(["icd_code", "icd_version"])
    rows_after_lk = clean_lookup.count()
    clean_lookup.write.mode("overwrite").parquet(PATHS["d_icd_diagnoses_clean"])

    append_report_section(
        f"{REPORTS_DIR}/cleaning_report.md", "d_icd_diagnoses",
        rows_before_lk, rows_after_lk, nulls_lk, dupes_lk,
    )

    print(f"diagnoses_icd: {rows_before_dx} -> {rows_after_dx}")
    print(f"d_icd_diagnoses: {rows_before_lk} -> {rows_after_lk}")
    spark.stop()


if __name__ == "__main__":
    main()
```

## C. W3 — Readmission Label Definition

### `src/labels/build_readmission_labels.py`

```python
"""
Builds the 30-day unplanned-readmission label on top of cleaned admissions.
Reads:  /data/mimic/cleaned/admissions
Writes: /data/mimic/cleaned/admissions_labeled

Run:
    spark-submit --master yarn --deploy-mode client src/labels/build_readmission_labels.py
"""
from pyspark.sql import Window, functions as F
from common.spark_session import get_spark
from common.config import PATHS


def main():
    spark = get_spark("build_readmission_labels")

    admissions = spark.read.parquet(PATHS["admissions_clean"])
    w = Window.partitionBy("subject_id").orderBy("admittime")

    labeled = (
        admissions
        .withColumn("next_admittime", F.lead("admittime").over(w))
        .withColumn("next_admission_type", F.lead("admission_type").over(w))
        .withColumn("days_to_next_admit", F.datediff(F.col("next_admittime"), F.col("dischtime")))
        .withColumn(
            "readmit_30d",
            F.when(
                (F.col("hospital_expire_flag") == 0)
                & F.col("days_to_next_admit").isNotNull()
                & (F.col("days_to_next_admit") <= 30)
                & (F.col("days_to_next_admit") >= 0)
                & (F.col("next_admission_type") != F.lit("ELECTIVE")),
                F.lit(1),
            ).otherwise(F.lit(0)),
        )
    )

    output = labeled.select(
        "subject_id", "hadm_id", "admittime", "dischtime", "los_days",
        "admission_type", "hospital_expire_flag", "days_to_next_admit", "readmit_30d",
    )
    output.write.mode("overwrite").parquet(PATHS["admissions_labeled"])

    total = output.count()
    positives = output.filter("readmit_30d = 1").count()
    print(f"admissions_labeled: {total} rows -> {PATHS['admissions_labeled']}")
    print(f"readmit_30d = 1: {positives} ({positives / total:.1%})")

    spark.stop()


if __name__ == "__main__":
    main()
```

"""
Sanity-checks the readmission label before anyone downstream trusts it.

Run:
    spark-submit --master yarn --deploy-mode client src/labels/validate_labels.py
"""
from common.spark_session import get_spark
from common.config import PATHS


def main():
    spark = get_spark("validate_labels")
    labeled = spark.read.parquet(PATHS["admissions_labeled"])

    print("== Class balance ==")
    labeled.groupBy("readmit_30d").count().show()

    print("== Sanity check: negative days_to_next_admit (should be 0) ==")
    bad = labeled.filter("days_to_next_admit < 0").count()
    print(f"Rows with negative days_to_next_admit: {bad}")
    assert bad == 0, "An admission's 'next' admission starts before this one's discharge — investigate."

    print("== Sanity check: hadm_id should be unique ==")
    dupes = labeled.groupBy("hadm_id").count().filter("count > 1").count()
    print(f"Duplicate hadm_ids: {dupes}")
    assert dupes == 0

    print("All checks passed.")
    spark.stop()


if __name__ == "__main__":
    main()
